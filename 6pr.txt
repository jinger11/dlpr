
# Import necessary libraries
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

from tensorflow.keras.applications import MobileNetV2
from tensorflow.keras.applications.mobilenet_v2 import preprocess_input
from tensorflow.keras.layers import Dense, GlobalAveragePooling2D, Dropout
from tensorflow.keras.models import Model

# A. Load CIFAR-10 dataset
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()

# Select only cat (3) and dog (5) images
train_mask = np.isin(y_train.flatten(), [3, 5])
test_mask = np.isin(y_test.flatten(), [3, 5])

x_train, y_train = x_train[train_mask], y_train[train_mask]
x_test, y_test = x_test[test_mask], y_test[test_mask]

# Convert labels: cat = 0, dog = 1
y_train = (y_train.flatten() == 5).astype('float32')
y_test = (y_test.flatten() == 5).astype('float32')

# Resize images to 96x96
x_train = tf.image.resize(x_train, (96, 96)).numpy()
x_test = tf.image.resize(x_test, (96, 96)).numpy()

# Preprocess images for MobileNetV2
x_train = preprocess_input(x_train)
x_test = preprocess_input(x_test)

# A. Load pre-trained MobileNetV2 model
base_model = MobileNetV2(
    weights='imagenet',
    include_top=False,
    input_shape=(96, 96, 3)
)

# B. Freeze lower convolutional layers
base_model.trainable = False

# C. Add custom classifier
x = base_model.output
x = GlobalAveragePooling2D()(x)
x = Dense(64, activation='relu')(x)
x = Dropout(0.3)(x)
output = Dense(1, activation='sigmoid')(x)

model = Model(inputs=base_model.input, outputs=output)

# Compile model
model.compile(
    optimizer='adam',
    loss='binary_crossentropy',
    metrics=['accuracy']
)

# D. Train custom classifier
history = model.fit(
    x_train, y_train,
    epochs=5,
    batch_size=32,
    validation_split=0.1
)

# Evaluate model
loss, accuracy = model.evaluate(x_test, y_test)
print("Test Accuracy:", accuracy * 100, "%")

# E. Fine-tune the upper layers
base_model.trainable = True

# Freeze all but the last 20 layers
for layer in base_model.layers[:-20]:
    layer.trainable = False

# Recompile with a small learning rate
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.00001),
    loss='binary_crossentropy',
    metrics=['accuracy']
)

# Train again for fine-tuning
model.fit(
    x_train, y_train,
    epochs=3,
    batch_size=32,
    validation_split=0.1
)

# Final evaluation
loss, accuracy = model.evaluate(x_test, y_test)
print("Final Test Accuracy:", accuracy * 100, "%")

# Plot training accuracy
plt.plot(history.history['accuracy'], label='Training Accuracy')
plt.plot(history.history['val_accuracy'], label='Validation Accuracy')
plt.xlabel('Epochs')
plt.ylabel('Accuracy')
plt.title('Transfer Learning Accuracy')
plt.legend()
plt.show()
