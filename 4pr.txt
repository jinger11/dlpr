
# Experiment 4: Autoencoder for Anomaly Detection

# a. Import required libraries
import numpy as np
import tensorflow as tf
from tensorflow.keras import layers, Model
import matplotlib.pyplot as plt

# b. Load and preprocess dataset
(x_train, _), (x_test, _) = tf.keras.datasets.mnist.load_data()

# Normalize pixel values between 0 and 1
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

# Flatten images from 28x28 to 784
x_train = x_train.reshape(-1, 784)
x_test = x_test.reshape(-1, 784)

# c. Build Encoder
input_layer = layers.Input(shape=(784,))

encoded = layers.Dense(128, activation="relu")(input_layer)
latent = layers.Dense(32, activation="relu")(encoded)

# d. Build Decoder
decoded = layers.Dense(128, activation="relu")(latent)
output_layer = layers.Dense(784, activation="sigmoid")(decoded)

# Create Autoencoder model
autoencoder = Model(input_layer, output_layer)

# e. Compile the model
autoencoder.compile(
    optimizer="adam",
    loss="mse",
    metrics=["mae"]
)

# Train the Autoencoder
history = autoencoder.fit(
    x_train,
    x_train,
    epochs=5,
    batch_size=256,
    validation_split=0.1
)

# Reconstruct test images
reconstructed = autoencoder.predict(x_test)

# Calculate test reconstruction error
errors = np.mean((x_test - reconstructed) ** 2, axis=1)

# Calculate training reconstruction error
train_reconstructed = autoencoder.predict(x_train)

train_errors = np.mean(
    (x_train - train_reconstructed) ** 2,
    axis=1
)

# Set anomaly threshold (95th percentile)
threshold = np.percentile(train_errors, 95)

# Detect potential anomalies
anomalies = errors > threshold

print("Reconstruction Threshold:", threshold)
print("Potential Anomalies:", np.sum(anomalies))
print("Test MAE:", np.mean(np.abs(x_test - reconstructed)))

# Plot training and validation loss
plt.figure(figsize=(8, 4))
plt.plot(history.history["loss"], label="Training Loss")
plt.plot(history.history["val_loss"], label="Validation Loss")
plt.xlabel("Epoch")
plt.ylabel("MSE")
plt.title("Autoencoder Training and Validation Loss")
plt.legend()
plt.tight_layout()
plt.show()

# Plot reconstruction error distribution
plt.figure(figsize=(8, 4))
plt.hist(train_errors, bins=50, alpha=0.6, label="Training Errors")
plt.hist(errors, bins=50, alpha=0.6, label="Test Errors")
plt.axvline(
    threshold,
    color="red",
    linestyle="--",
    label="Anomaly Threshold"
)
plt.xlabel("Reconstruction MSE")
plt.ylabel("Number of Images")
plt.title("Reconstruction Error Distribution")
plt.legend()
plt.tight_layout()
plt.show()

# Display original and reconstructed images
fig, axes = plt.subplots(2, 8, figsize=(12, 4))

for i in range(8):
    axes[0, i].imshow(
        x_test[i].reshape(28, 28),
        cmap="gray"
    )
    axes[1, i].imshow(
        reconstructed[i].reshape(28, 28),
        cmap="gray"
    )

    axes[0, i].axis("off")
    axes[1, i].axis("off")

axes[0, 0].set_ylabel("Original")
axes[1, 0].set_ylabel("Reconstructed")

fig.suptitle("Original and Reconstructed MNIST Images")
plt.tight_layout()
plt.show()
